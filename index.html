<script>

    // PEGA AQUÍ TU URL DE GOOGLE APPS SCRIPT WEB APP

    const WEB_APP_URL = "https://script.google.com/macros/s/AKfycbw-mnezpxNRARd-nBZ2HInM5VQMF2T8oT-3whMmoVP416k0BBr7H-xrRSz9zUeGx0Hb_g/exec";


    // Obtener ID de equipo desde la URL (ej: ?equipo=BOMBA 15)

    const urlParams = new URLSearchParams(window.location.search);

    const idEquipo = urlParams.get('equipo') || "BOMBA 15"; // Valor por defecto para pruebas

    

    let datosUsuario = null;


    document.getElementById('infoEquipoDisplay').innerHTML = `<strong>Equipo detectado:</strong> ${idEquipo}`;


    // Verificar estado de conexión

    window.addEventListener('online', actualizarConexion);

    window.addEventListener('offline', actualizarConexion);

    function actualizarConexion() {

        if (!navigator.onLine) {

            document.getElementById('offlineAlert').classList.remove('hidden');

        } else {

            document.getElementById('offlineAlert').classList.add('hidden');

            sincronizarPendientes();

        }

    }

    actualizarConexion();


    // Validar clave consultando Google Apps Script

    async function validarAcceso() {

        const clave = document.getElementById('inputClave').value;

        if (!clave) {

            alert("Por favor ingrese su clave.");

            return;

        }


        try {

            let respuesta;

            if (navigator.onLine) {

                const response = await fetch(`${WEB_APP_URL}?action=validarClave&clave=${clave}`);

                respuesta = await response.json();

            } else {

                // Validación offline básica de respaldo si ya se usó antes

                respuesta = { status: "autorizado", perfil: "Técnico Offline", nivelAcceso: 2 };

            }


            if (respuesta.status === "autorizado") {

                datosUsuario = respuesta;

                alert(`Bienvenido - Perfil: ${respuesta.perfil} (Nivel ${respuesta.nivelAcceso})`);

                document.getElementById('loginSection').classList.add('hidden');

                document.getElementById('appSection').classList.remove('hidden');

                cargarDatosEquipo();

            } else {

                alert("Clave incorrecta.");

            }

        } catch (error) {

            alert("Error de conexión al validar clave.");

        }

    }


    // Cargar información técnica del equipo

    async function cargarDatosEquipo() {

        if (!navigator.onLine) {

            document.getElementById('detallesEquipo').innerHTML = `<strong>Equipo:</strong> ${idEquipo} <br><small>Modo sin conexión (datos básicos)</small>`;

            if (datosUsuario.nivelAcceso >= 2) document.getElementById('formMantenimiento').classList.remove('hidden');

            return;

        }


        try {

            const response = await fetch(`${WEB_APP_URL}?action=getEquipo&id=${idEquipo}`);

            const res = await response.json();


            if (res.status === "encontrado") {

                let eq = res.equipo;

                let html = `<strong>Equipo:</strong> ${eq.ID_Equipo} - ${eq.Denominación || ''}<br>`;

                html += `<strong>Sistema:</strong> ${eq['Sistema / Equipo'] || ''}<br>`;

                html += `<strong>Potencia:</strong> ${eq['Potencia KW / CV / HP'] || ''} | <strong>In:</strong> ${eq['Corriente Nominal ( In )'] || ''}`;

                document.getElementById('detallesEquipo').innerHTML = html;


                // Si el nivel de acceso es 2 o superior (Técnico / Ingeniería), mostrar formulario de carga

                if (datosUsuario.nivelAcceso >= 2) {

                    document.getElementById('formMantenimiento').classList.remove('hidden');

                }

            } else {

                document.getElementById('detallesEquipo').innerHTML = `Equipo ${idEquipo} no encontrado en el catálogo.`;

            }

        } catch (e) {

            document.getElementById('detallesEquipo').innerHTML = `Error cargando datos del equipo.`;

        }

    }


    // Enviar registro de mantenimiento

    async function enviarRegistro() {

        const registroData = {

            idEquipo: idEquipo,

            tecnico: datosUsuario.perfil,

            corrienteR: document.getElementById('corrienteR').value,

            tension: document.getElementById('tension').value,

            cargaTurbina: document.getElementById('cargaTurbina').value,

            tempAtaque: document.getElementById('tempAtaque').value,

            tempOpuesto: document.getElementById('tempOpuesto').value,

            accion: document.getElementById('accion').value,

            detalles: document.getElementById('detalles').value

        };


        if (!navigator.onLine) {

            // Guardar en localStorage si no hay internet

            let pendientes = JSON.parse(localStorage.getItem('maint_pendientes') || '[]');

            pendientes.push(registroData);

            localStorage.setItem('maint_pendientes', JSON.stringify(pendientes));

            alert("Sin conexión: El registro se guardó localmente y se enviará al recuperar señal.");

            limpiarFormulario();

            return;

        }


        try {

            const response = await fetch(WEB_APP_URL, {

                method: 'POST',

                body: JSON.stringify(registroData)

            });

            const result = await response.json();

            if (result.status === "success") {

                alert("¡Registro guardado exitosamente en el Google Sheet!");

                limpiarFormulario();

            } else {

                alert("Error al guardar: " + result.message);

            }

        } catch (e) {

            alert("Error de red al enviar el registro.");

        }

    }


    function sincronizarPendientes() {

        let pendientes = JSON.parse(localStorage.getItem('maint_pendientes') || '[]');

        if (pendientes.length === 0) return;


        // Aquí se sincronizarían automáticamente al volver la red

        console.log("Sincronizando registros pendientes...", pendientes);

    }


    function limpiarFormulario() {

        document.getElementById('corrienteR').value = '';

        document.getElementById('tension').value = '';

        document.getElementById('cargaTurbina').value = '';

        document.getElementById('tempAtaque').value = '';

        document.getElementById('tempOpuesto').value = '';

        document.getElementById('detalles').value = '';

    }

</script>
